# 刷草（植物生成）相关代码完整整理

刷草（植物生成）逻辑完整链路如下，涉及 6 个文件。核心在 `game\_normal.cjs`，残局/枯萎模式在 `game\_blight.cjs` 中被关闭。

## 1\. 入口调用点 — `js/global/fascend/runtime.cjs`（第 336 行）

每回合落子结算完、清场前调用：


### 4.2 `ed\_pl\_1()` — 生成植物主入口（第 168-252 行）

```javascript
/\*\*
 \* @description: 5. 生成植物
 \* @return {Pos\[] | void}
 \*/
ed\_pl\_1() {
    if (this.state.ascend\_status === ASCEND\_ENUM.START) return;  // ASCEND发起后、回应前不刷草
    if (!this.natural\_plant) return;                             // 残局等模式关闭自然刷草


    const atk = this.state.ascend\_status === ASCEND\_ENUM.END;    // 本回合是否完成了完整ASCEND

    // 非完整ASCEND回合：推进反升变蓄力冷却
    if (!atk) {
        if (this.state.unascend\_charge > 0) this.state.unascend\_charge--;

        if (this.state.unascend\_chargef\[1] > 0) this.state.unascend\_chargef\[1]--;
        if (this.state.unascend\_chargef\[0] > 0) this.state.unascend\_chargef\[0]--;

        this.state.just\_unascend = false;
    }


    this.state.grow\_count--;
    // 冷却未到：直接返回；但完整ASCEND且非just\_unascend时强制刷一次
    if (this.state.grow\_count > 0 \&\& !(atk \&\& !this.state.just\_unascend)) return;

    const self\_index = this.state.turn\_player === PLAYER\_ENUM.BLACK ? 0 : 1;
    const foe\_index = this.state.turn\_player === PLAYER\_ENUM.BLACK ? 1 : 0;

    const stone\_count = this.board.stone\_count;
    const plant\_count = this.state.turn\_count >= 65 ? 3 : 2;   // 65回合后每次刷3株，否则2株


    // 最低7回合，最大11回合，每22回合减少2最大回合数（11）
    this.state.grow\_count = Math.max(7, 11 - Math.trunc(Math.trunc(this.state.turn\_count / 22) \* 2));


    // 超载判定：永夜模式跳过棋子数量和血量条件，始终直接执行超载
    if (this.always\_overload) this.state.over\_status = OVER\_ENUM.EXECUTE;
    else {
        // 44子则超载，若我方血量少于或者等于敌方，进入超载EX
        // 22子以下，则取消所有超载状态
        if (stone\_count >= 44 \&\& this.state.over\_status === OVER\_ENUM.NONE) this.state.over\_status = OVER\_ENUM.ENABLE;
        if (this.state.over\_status === OVER\_ENUM.ENABLE \&\& this.state.hp\[self\_index] <= this.state.hp\[foe\_index]) this.state.over\_status = OVER\_ENUM.EXECUTE;
        if (this.state.over\_status !== OVER\_ENUM.NONE \&\& stone\_count < 22) this.state.over\_status = OVER\_ENUM.NONE;
    }


    // 生成草（扫描→计分→筛选）
    const result = this.p\_create(plant\_count);

    // 创建草：空格点长一草，已有一草的升为二草
    if (result) result.forEach((pos) => {
        const plant = pos.plant === PLANT\_ENUM.NONE ? PLANT\_ENUM.ONE : PLANT\_ENUM.TWO;
        this.board.set\_plant(pos.x, pos.y, plant);
    });


    // 若超载，减少50%冷却
    // 若冷却为偶数，减少1，变为奇数（保证轮流生成草）
    if (this.state.over\_status !== OVER\_ENUM.NONE) this.state.grow\_count = Math.trunc(this.state.grow\_count / 2);
    if (this.state.grow\_count % 2 == 0) this.state.grow\_count--;


    // 完整ASCEND后改变数据
    if (atk) {
        // 反升变蓄力按 zsin 曲线回收：12.5 \~ 25 之间渐进回升
        this.state.unascend\_charge = Math.ceil(AIC.mmx(
            12.5,
            this.state.unascend\_charge + (25 - this.state.unascend\_charge) \* 0.4,
            25
        ));

        // 完整ASCEND方获得快速反升变冷却：超载期4回合，平时9回合
        this.state.unascend\_chargef\[self\_index] = Math.max(
            this.state.unascend\_chargef\[self\_index],
            this.state.over\_status !== OVER\_ENUM.NONE ? 4 : 9
        );

        this.state.just\_unascend = true;
    }

    return result;
};
```

### 4.3 `p\_create()` — 生成流程编排（第 507-544 行）

```javascript
/\*\*
 \* @description: 生成植物
 \* @param {number} count
 \* @return {Pos\[] | void}
 \*/
p\_create(count) {
    const self\_index = this.state.turn\_player === PLAYER\_ENUM.BLACK ? 0 : 1;
    const foe\_index = this.state.turn\_player === PLAYER\_ENUM.BLACK ? 1 : 0;

    let atk = this.state.ascend\_status === ASCEND\_ENUM.END; // 临时

    const flag = this.state.unascend\_chargef\[self\_index] > 0;    // 我方处于快速反升变冷却
    const flag2 = this.state.unascend\_chargef\[foe\_index] > 0;    // 敌方处于快速反升变冷却
    const flag3 = atk;                                           // 完整ASCEND：扫描含ASCEND用子自身
    const flag4 = atk \&\& this.state.unascend\_charge <= 0;        // 完整ASCEND且全局蓄力耗尽


    if (this.state.unascend\_chargef\[self\_index] > 0) atk = false; // 我方冷却中则视为非攻击刷草


    // 扫描符合规则点位
    const cand = this.p\_scan\_pos(flag3, flag4);

    // 排序（对齐AIC扫描顺序）
    this.p\_sort\_pos(cand);

    // 若超载EX，增加1数量
    // 取最小数量
    // 若符合点位数量为0，则不进行任何生成
    if (this.state.over\_status === OVER\_ENUM.EXECUTE) count++;
    count = Math.min(count, cand.length);
    if (count <= 0) return;

    // 对点位计分，筛选，得到最终点位
    const result = this.p\_scan\_score(cand, atk, count, flag, flag2, flag4);

    return result;
};
```

### 4.4 `p\_scan\_pos()` — 扫描合法生成点（第 254-310 行）

```javascript
/\*\*

 \* @param {boolean} flag3   是否完整ASCEND回合
 \* @param {boolean} flag4   是否完整ASCEND且全局蓄力耗尽

 \*/
p\_scan\_pos(flag3, flag4) {
    const tst = \[];

    // 若条件满足，则第一轮扫描计算满足一方的ASCEND用子（允许双方都满足）
    // 若双方条件都不满足，则仅计算填充子

    if (this.state.unascend\_chargef\[1] === 0) tst.push(STONE\_ENUM.WHITE\_USED);
    if (tst.length === 0) tst.push(STONE\_ENUM.BLACK, STONE\_ENUM.WHITE);


    const result = \[];
    /\*\*
     \* @param {Pos} pos
     \*/
    const func = (pos) => {
        const filled = pos.stone === STONE\_ENUM.BLACK || pos.stone === STONE\_ENUM.WHITE;

        // flag3模式：判定该子是否属于可用ASCEND用子（或超载期填充子）
        const a = tst.includes(pos.stone) || (filled \&\& (flag4 || this.state.over\_status !== OVER\_ENUM.NONE));
        const b = filled;

        ds.push(...s);

        ds.forEach((p) => {

            result.push(p);
        });
    };


    // 默认进行第一次扫描
    this.board.data.forEach(func);

    // 若没有没有扫描到符合点位或完整ASCEND，则扫描第二轮
    // 第二轮扫描默认计算双方的ASCEND用子
    if (result.length > 0 \&\& flag3) return result;

    tst.push(STONE\_ENUM.BLACK\_USED, STONE\_ENUM.WHITE\_USED);
```

### 4.5 `p\_sort\_pos()` — 强制扫描顺序（第 494-505 行）

```javascript
/\*\*
 \* @description: 对扫描到的符合规则的点位排序
 \* @param {Pos\[]} points
 \* @return {void}
 \*/
p\_sort\_pos(points) {
    // 强制排序为AIC实现扫描的顺序！
    points.sort((a, b) => {
        if (a.y !== b.y) return a.y - b.y; // Y 升序 (0, 1, 2...)
        return a.x - b.x;                 // X 升序 (0, 1, 2...)
    });
};
```

### 4.6 `p\_scan\_score()` — 计分筛选（第 312-492 行）

```javascript
/\*\*
 \* @description: 对生成点计分且筛选
 \* @param {Pos\[]} points
 \* @param {boolean} atk     是否攻击型刷草（完整ASCEND后）
 \* @param {number} count    计划生成数量
 \* @param {boolean} flag    我方快速反升变冷却中
 \* @param {boolean} flag2   敌方快速反升变冷却中
 \* @param {boolean} flag4   完整ASCEND且全局蓄力耗尽
 \* @return {Pos\[]}
 \*/
p\_scan\_score(points, atk, count, flag, flag2, flag4) {
    const FS\_OVER = false;

    const tst = \[];

    const self\_index = this.state.turn\_player === PLAYER\_ENUM.BLACK ? 0 : 1;
    const foe\_index = this.state.turn\_player === PLAYER\_ENUM.BLACK ? 1 : 0;


    // 若条件满足，则第一轮扫描计算满足一方的ASCEND用子（允许双方都满足）
    // 若双方条件都不满足，则仅计算填充子
    if (this.state.unascend\_chargef\[0] === 0) tst.push(STONE\_ENUM.BLACK\_USED);
    if (this.state.unascend\_chargef\[1] === 0) tst.push(STONE\_ENUM.WHITE\_USED);
    if (tst.length === 0) tst.push(STONE\_ENUM.BLACK, STONE\_ENUM.WHITE);


    const result = \[];
    const buf = \[];

    let \_byte = 0;

    points.forEach((pos) => {
        let score = 500;

        let flag5 = false;
        let flag6 = atk || FS\_OVER;

        let abyte = 0;


        const self\_stone = this.state.turn\_player === PLAYER\_ENUM.BLACK ? STONE\_ENUM.BLACK : STONE\_ENUM.WHITE;
        const foe\_stone = this.state.turn\_player === PLAYER\_ENUM.BLACK ? STONE\_ENUM.WHITE : STONE\_ENUM.BLACK;

        const \[self\_align, self\_align\_t] = this.board.get\_align(pos.x, pos.y, self\_stone);
        const \[foe\_align, foe\_align\_t] = this.board.get\_align(pos.x, pos.y, foe\_stone);


        // 任一方连线≥4的点位：超载执行期进buf按bit位补位，并扣分（避免直接喂给成打）
        if (self\_align >= 4 || foe\_align >= 4) {
            if (this.state.over\_status === OVER\_ENUM.EXECUTE) {
                let byte = 0;

                if (self\_align < 4 \&\& !flag2) byte |= 2;   // bit2：利于敌方方向
                if (foe\_align < 4 \&\& !flag) byte |= 1;     // bit1：利于我方方向

                if (byte == 1 || byte == 2) {
                    score += 450;
                    if (pos.plant === PLANT\_ENUM.NONE) score += 100;

                    if (byte == 1 || FS\_OVER) flag6 = true;
                    abyte = byte;
                }
            }
            score -= 450;
        }


        if (flag6) {
            // 攻击型：偏向敌我连线差小的均衡点
            if (self\_align\_t < foe\_align\_t) score -= Math.abs(self\_align\_t - foe\_align\_t) \* 3;
            if (self\_align < foe\_align) score -= Math.abs(self\_align - foe\_align) \* 15;

            // 我方3连且敌方≤1：按反升变蓄力正弦曲线加分（蓄力越低加成越高）；蓄力耗尽则标记flag5双份候选
            if (self\_align === 3 \&\& foe\_align <= 1) {
                if (flag4) flag5 = true;
                else score += Math.trunc(AIC.zsin(25 - this.state.unascend\_charge, 25) \* 120);
            }

        } else {
            // 非攻击型：偏向冷却方连线占优的点
            if (self\_align\_t < foe\_align\_t \&\& flag \&\& !flag2) score += Math.abs(self\_align\_t - foe\_align\_t) \* 6;
            else if (self\_align\_t > foe\_align\_t \&\& !flag \&\& flag2) score += Math.abs(self\_align\_t - foe\_align\_t) \* 6;
            else if (self\_align\_t !== foe\_align\_t) score -= Math.abs(self\_align\_t - foe\_align\_t) \* 3;

            if (self\_align !== foe\_align) score -= Math.abs(self\_align - foe\_align) \* 15;
        }


        // 已有草的非耗尽场景扣分；攻击型刷草落在ASCEND用子上加分
        if (pos.plant !== PLANT\_ENUM.NONE \&\& !flag4) score -= 30;
        else if (atk \&\& tst.includes(pos.stone)) score += 50;


        score += this.rand\_aic.xor(20);   // 随机扰动 0\~19


        if (abyte > 0) {
            \_byte |= abyte;

            buf.push({
                'score': score,
                'bit': abyte,
                'pos': pos
            });

        } else {
            result.push({
                'score': score,
                'bit': 0,
                'pos': pos
            });

            // flag5点额外塞入一个略低分副本，提高中选概率
            if (flag5) {
                count++;
                result.push({
                    'score': score + 30 - this.rand\_aic.xor(90),
                    'bit': 0,
                    'pos': pos
                });
            }
        }
    });


    // 合并到最终：超载补位点按bit位需求洗牌后注入（每点注入原分+扰动副本两份）
    if (\_byte > 0) {
        this.rand\_unity.shuffle(buf);

        const count2 = buf.length;
        let num7 = 0;

        while (num7 < count2 \&\& \_byte > 0) {
            const pt = buf\[num7];
            const bit = pt.bit;

            if ((\_byte \& bit) != 0) {
                \_byte \&= \~bit;
                count++;

                const pt2 = { ...pt };
                pt2.score += 40 - this.rand\_aic.xor(120);

                result.push(pt);
                result.push(pt2);
            }
            num7++;
        }
    }


    // 生成植物
    // 倒序排序！！！
    // 注意是倒序，从大到小！
    this.rand\_unity.shuffle(result);
    CsharpSort.sort(result, 0, result.length, (a, b) => b.score - a.score);

    let num9 = 0;
    count = Math.min(count, 6);   // 单回合最多刷6个位置

    const poses = \[];
    const fix = \[];

    while (num9 < result.length \&\& count > 0) {
        const pt = result\[num9];
        const pos = pt.pos;
        const bit = pt.bit;

        // 同一位置只允许中选一次（一草点进入fix禁止其二草副本再选）
        if (!fix.includes(pos)) {
            poses.push(pos);
            if (pos.plant === PLANT\_ENUM.ONE) fix.push(pos);

            count--;

            // 补位点同时给对应方施加4回合快速反升变冷却
            if ((bit \& 1) != 0) this.state.unascend\_chargef\[self\_index] = Math.max(this.state.unascend\_chargef\[self\_index], 4);
            if ((bit \& 2) != 0) this.state.unascend\_chargef\[foe\_index] = Math.max(this.state.unascend\_chargef\[foe\_index], 4);
        }
        num9++;
    }

    return poses;
};
```

```javascript
        \[-1, 1],  // 左上
        \[-1, -1], // 左下

        \[1, 1],   // 右上
        \[1, -1],  // 右下
    ]; // 警告，为了统一AIC访问顺序，特此定义

    const result = \[];

    directions.forEach((\[dx, dy]) => {
        const sx = x + dx;
        const sy = y + dy;
        if (!this.in\_bounds(sx, sy)) return;

        const p = this.data\[this.xy2index(sx, sy)];
        result.push(p);
    });

    return result;
};

/\*\*
 \* @description 获取 max\_align max\_align\_total || 最大 总和
 \* @param {number} x
 \* @param {number} y
 \* @param {STONE\_ENUM} stone
 \* @returns {\[number, number]}
 \*/
get\_align(x, y, stone) {
    const directions = \[
        \[0, -1],  // 上
        \[0, 1],   // 下

        \[-1, 0],  // 左
        \[1, 0],   // 右

        \[-1, -1], // 左上
        \[1, 1],   // 右下

        \[-1, 1],  // 左下
        \[1, -1],  // 右上
    ];

    const result = \[];
    for (const \[dx, dy] of directions) {
        // 必须连续，不允许断开或者敌方棋子
        // 中心默认计数1，就算非我方棋子也计数
        let c = 1;

        let sx = x;
        let sy = y;

        sx += dx;
        sy += dy;

        while (this.in\_bounds(sx, sy)) {
            if (this.get\_stone(sx, sy) !== stone) break;

            c++;
            sx += dx;
            sy += dy;
        }

        result.push(c);
    }

    // 以两组，计算连线
    const line = \[];
    for (let index = 0; index < 8; index += 2) {
        const a = result\[index] - 1;
        const b = result\[index + 1] - 1;

        line.push(a + b + 1); // 中心+1
    }

    // 取最大连线长度
    const max\_align = Math.max(...line);

    // 计算所有方向总连续长度
    let max\_align\_total = -7; // 八方向，默认-8，但中心+1
    result.forEach((c) => max\_align\_total += c);

    return \[max\_align, max\_align\_total];
};
```
